set -e
# Called in $(...), so the message goes to stderr and set -e stops the script on failure.
inside() {
  r=$(realpath -m -- "$1") || { echo "ERROR: invalid path" >&2; exit 3; }
  case "$r" in
    /var/www/gamedata|/var/www/gamedata/*) printf '%s' "$r" ;;
    *) echo "ERROR: that path is outside /var/www/gamedata" >&2; exit 3 ;;
  esac
}
# The web server and the emulator, both in group www-data, may read and change uploaded files.
fix_rights() {
  chown -R www-data:www-data -- "$1"
  find "$1" -type d -exec chmod 2775 {} +
  find "$1" -type f -exec chmod 664 {} +
}
set -e
d=$(inside '/var/www/gamedata')
[ -d "$d" ] || { echo "ERROR: the target is not a folder"; exit 3; }
list=$(mktemp)
trap 'rm -f "$list" "$list.tops"' EXIT
tar -x -v --quoting-style=literal -f - -C "$d" --no-same-owner --no-same-permissions --skip-old-files > "$list"
cut -d/ -f1 "$list" | sort -u | while IFS= read -r top; do
  case "$top" in ''|.|..) continue ;; esac
  if [ -e "$d/$top" ] && [ ! -L "$d/$top" ]; then printf '%s\0' "$d/$top"; fi
done > "$list.tops"
if [ -s "$list.tops" ]; then
  xargs -0 chown -R www-data:www-data -- < "$list.tops"
  xargs -0 sh -c 'find "$@" -type d -exec chmod 2775 {} + && find "$@" -type f -exec chmod 664 {} +' fix_rights < "$list.tops"
fi
echo "  $(wc -l < "$list") item(s) in $d" >&2
echo OK
