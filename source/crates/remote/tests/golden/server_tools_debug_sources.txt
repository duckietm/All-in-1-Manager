### Firewall

/var/log/ufw.log (or the journal without rsyslog)
setsid bash -c 'if [ -f /var/log/ufw.log ]; then tail -n 200 -F /var/log/ufw.log; else echo '\''No ufw.log (no rsyslog): following the journal instead.'\''; journalctl -k -f -n 200 -o short-iso | grep --line-buffered UFW; fi' &

### Fail2Ban

/var/log/fail2ban.log (or the journal)
setsid bash -c 'if ! command -v fail2ban-client >/dev/null 2>&1; then echo '\''Fail2ban is not installed on this server (Addons -> Fail2Ban).'\''; sleep infinity; elif [ -f /var/log/fail2ban.log ]; then tail -n 200 -F /var/log/fail2ban.log; else echo '\''No fail2ban.log: following the journal instead.'\''; journalctl -u fail2ban -f -n 200 -o short-iso; fi' &
follower=$!
wait $ALL_IN_1_WATCH
kill -- -$follower 2>/dev/null
true
