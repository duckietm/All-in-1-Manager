# docker compose in a folder, with the owner's own compose.override.yaml when there is one (it survives reruns).
compose_in() {
  if [ -f "$1/compose.override.yaml" ]; then docker compose -f "$1/compose.yaml" -f "$1/compose.override.yaml" "${@:2}"
  else docker compose -f "$1/compose.yaml" "${@:2}"; fi
}

# apt refuses repository files dated in the future ("Release file ... is not valid yet"), so a server
# clock that runs behind breaks every apt-get update. Compare it with the Date header of a trusted
# HTTPS server and correct it once, before the first update.
clock_checked=
fix_clock() {
  [ -n "$clock_checked" ] && return 0
  clock_checked=1
  remote=""
  for url in https://deb.debian.org/ https://archive.ubuntu.com/; do
    if command -v curl >/dev/null 2>&1; then
      remote=$(curl -fsSI --max-time 10 "$url" 2>/dev/null | tr -d '\r' | awk 'tolower($1)=="date:" {sub(/^[^:]*: */, ""); print; exit}')
    elif command -v wget >/dev/null 2>&1; then
      remote=$(wget -qS --spider -T 10 "$url" 2>&1 | tr -d '\r' | awk 'tolower($1)=="date:" {sub(/^[^:]*: */, ""); print; exit}')
    fi
    [ -n "$remote" ] && break
  done
  [ -n "$remote" ] || { echo "  (could not check the server clock: no curl/wget or no internet)"; return 0; }
  real=$(date -d "$remote" +%s 2>/dev/null) || return 0
  skew=$((real - $(date +%s)))
  if [ "${skew#-}" -gt 120 ]; then
    echo "  The server clock is off by $skew seconds: setting it to $remote"
    date -u -s "@$real" >/dev/null || fail "The server clock is off by $skew seconds and could not be corrected. Set the right time, then try again."
    hwclock --systohc >/dev/null 2>&1 || true
    if command -v timedatectl >/dev/null 2>&1 && [ -d /run/systemd/system ] && timedatectl set-ntp true >/dev/null 2>&1; then
      echo "  Automatic time sync (NTP) is on"
    else
      echo "  Automatic time sync could not be switched on: install systemd-timesyncd or chrony to keep the time right"
    fi
  fi
}
apt_update() { fix_clock; apt-get update; }

# Another apt holds its locks (Ubuntu's unattended-upgrades runs right after an install or a boot, apt-daily later):
# without waiting, apt-get stops with "Could not get lock /var/lib/dpkg/lock-frontend". The locks are read from
# /proc/locks (by inode), not guessed from process names. DPkg::Lock::Timeout covers one starting just after the check.
apt_locked() {
  for lock in /var/lib/dpkg/lock-frontend /var/lib/dpkg/lock /var/lib/apt/lists/lock /var/cache/apt/archives/lock; do
    [ -e "$lock" ] || continue
    awk -v inode="$(stat -c %i "$lock")" '{ split($6, id, ":"); if (id[3] == inode) found = 1 } END { exit !found }' /proc/locks 2>/dev/null && return 0
  done
  return 1
}
wait_for_apt() {
  apt_locked || return 0
  echo "  Another apt is busy (often Ubuntu's automatic updates after a boot): waiting until it is done ..."
  waited=0
  while apt_locked; do
    [ "$waited" -ge 1800 ] && fail "apt is still busy after 30 minutes (another process holds its lock). Wait until it is done, or reboot the server, then try again."
    sleep 5
    waited=$((waited + 5))
    [ $((waited % 60)) -eq 0 ] && echo "  still waiting ($((waited / 60)) min)"
  done
  echo "  apt is free again"
}
apt-get() { wait_for_apt; command apt-get -o DPkg::Lock::Timeout=600 "$@"; }
dpkg() {
  case "$1" in -i|--install|--configure|-r|--remove|-P|--purge|--unpack) wait_for_apt ;; esac
  command dpkg "$@"
}

# download URL FILE, with timeouts and retries and a last attempt over IPv4. Tools that download by
# themselves (composer-setup.php, mvnw) have no timeout and hang forever on a broken IPv6 route.
download() {
  curl -fsSL --connect-timeout 20 --max-time 900 --retry 2 --retry-delay 5 -o "$2" "$1" && return 0
  echo "  Trying again over IPv4 ..."
  curl -fsSL -4 --connect-timeout 20 --max-time 900 --retry 2 --retry-delay 5 -o "$2" "$1"
}

# Opens ports when ufw is active, unless the harden step manages the firewall.
ufw_allow() {
  if [ -f /etc/ufw/all-in-1-hardened ]; then
    echo "  The firewall is managed by the Harden step: no port opened (run that step again to change it)"
    return 0
  fi
  command -v ufw >/dev/null 2>&1 && ufw status | grep -q 'Status: active' || return 0
  echo "> ufw is active: allowing $*"
  for port in "$@"; do ufw allow "$port"; done
}

# Sets $ranges to Cloudflare's published IP ranges, one per line. $1 is added to the error message.
load_cloudflare_ranges() {
  fix_clock
  echo "> Downloading Cloudflare's IP ranges (www.cloudflare.com/ips-v4, ips-v6)"
  ranges=$( (curl -fsS --max-time 20 https://www.cloudflare.com/ips-v4; echo; curl -fsS --max-time 20 https://www.cloudflare.com/ips-v6) \
    | grep -E '^[0-9a-f.:]+/[0-9]+$' ) || true
  [ "$(echo "$ranges" | grep -c .)" -ge 10 ] || fail "Could not download Cloudflare's IP ranges${1:+; $1}."
  echo "  $(echo "$ranges" | grep -c .) ranges"
}

-- ---------------------------------------------------------------- furniture
-- One Plus definition per classname and type (the one with furnidata, else the lowest id).
DROP TABLE IF EXISTS _pi_furni;
CREATE TABLE _pi_furni (
  plus_id INT UNSIGNED PRIMARY KEY,
  name_key VARCHAR(100) COLLATE utf8mb4_general_ci NOT NULL,
  type CHAR(1) NOT NULL,
  canonical INT UNSIGNED NOT NULL,
  polaris_id INT UNSIGNED NULL,
  is_new TINYINT NOT NULL DEFAULT 0,
  interaction VARCHAR(100) COLLATE utf8mb4_general_ci NULL,
  how VARCHAR(20) NULL,
  KEY (name_key, type), KEY (canonical)
) ENGINE=InnoDB;
INSERT INTO _pi_furni (plus_id, name_key, type, canonical)
SELECT f.id, LOWER(TRIM(REGEXP_REPLACE(f.item_name, '[[:cntrl:]]', '') COLLATE utf8mb4_general_ci)), f.type,
       FIRST_VALUE(f.id) OVER (PARTITION BY LOWER(TRIM(REGEXP_REPLACE(f.item_name, '[[:cntrl:]]', '') COLLATE utf8mb4_general_ci)), f.type ORDER BY f.has_furnidata DESC, f.id)
FROM furniture f;

-- Polaris already has it: keep its id and its definition.
DROP TABLE IF EXISTS _pi_existing;
CREATE TABLE _pi_existing (name_key VARCHAR(100) COLLATE utf8mb4_general_ci, type VARCHAR(3), id INT UNSIGNED, PRIMARY KEY (name_key, type)) ENGINE=InnoDB
SELECT LOWER(TRIM(CONVERT(item_name USING utf8mb4) COLLATE utf8mb4_general_ci)) AS name_key, type, MIN(id) AS id FROM `@T@`.items_base GROUP BY 1, 2;
UPDATE _pi_furni p JOIN _pi_existing e ON e.name_key = p.name_key AND e.type = p.type SET p.polaris_id = e.id;

-- New: the next ids after the highest one Polaris has.
SET @base = (SELECT COALESCE(MAX(id), 0) FROM `@T@`.items_base);
DROP TABLE IF EXISTS _pi_new_ids;
CREATE TABLE _pi_new_ids (canonical INT UNSIGNED PRIMARY KEY, polaris_id INT UNSIGNED) ENGINE=InnoDB
SELECT canonical, @base + ROW_NUMBER() OVER (ORDER BY canonical) AS polaris_id
FROM (SELECT DISTINCT canonical FROM _pi_furni WHERE polaris_id IS NULL) c;
UPDATE _pi_furni p JOIN _pi_new_ids n ON n.canonical = p.canonical SET p.polaris_id = n.polaris_id, p.is_new = 1;

-- The interaction of a new definition, first rule that gives a type Polaris knows.
UPDATE _pi_furni p JOIN furniture f ON f.id = p.plus_id
LEFT JOIN _pi_by_name b ON b.item_name = p.name_key
LEFT JOIN _pi_known kb ON kb.name = b.interaction
LEFT JOIN _pi_known kn ON kn.name = p.name_key
  AND (f.interaction_type LIKE 'wired%' OR f.interaction_type LIKE 'wf\_%' OR p.name_key LIKE 'wf\_%' OR p.name_key LIKE 'conf\_%')
LEFT JOIN _pi_reverse r ON r.plus = LOWER(f.interaction_type COLLATE utf8mb4_general_ci)
LEFT JOIN _pi_known kr ON kr.name = r.polaris
LEFT JOIN _pi_known ki ON ki.name = LOWER(f.interaction_type COLLATE utf8mb4_general_ci)
SET p.interaction = COALESCE(kb.name, kn.name, kr.name, ki.name, 'default'),
    p.how = CASE WHEN kb.name IS NOT NULL THEN 'classname' WHEN kn.name IS NOT NULL THEN 'wired' WHEN kr.name IS NOT NULL THEN 'renamed'
                 WHEN ki.name IS NOT NULL THEN 'same' WHEN LOWER(f.interaction_type COLLATE utf8mb4_general_ci) = 'default' THEN 'same' ELSE 'fallback' END
WHERE p.is_new = 1 AND p.plus_id = p.canonical;

SELECT id AS plus_id, ROW_NUMBER() OVER (ORDER BY id) AS polaris_id,
       CASE layout WHEN 'frontpage4' THEN 'frontpage' WHEN 'club_gifts' THEN 'club_gift' WHEN 'guild_frontpage' THEN 'guilds'
                   WHEN 'guild_custom_furni' THEN 'guild_furni' WHEN 'default_3x3_extrainfo' THEN 'default_3x3' ELSE layout END AS layout

UPDATE _pi_pages SET layout = 'default_3x3'
WHERE layout NOT IN (SELECT TRIM(BOTH '''' FROM j.v) FROM (
  SELECT SUBSTRING_INDEX(SUBSTRING_INDEX(SUBSTRING(column_type, 6, CHAR_LENGTH(column_type) - 6), ',', n.n), ',', -1) AS v
  FROM information_schema.columns
  JOIN (SELECT seq AS n FROM seq_1_to_100) n
  WHERE table_schema = '@T@' AND table_name = 'catalog_pages' AND column_name = 'page_layout'
    AND n.n <= 1 + CHAR_LENGTH(column_type) - CHAR_LENGTH(REPLACE(column_type, ',', ''))) j);

-- Text and image slots, as Polaris's layouts read them (the reverse of plus-migrator's slots()). Every page's slots
-- in one row first, so nothing is looked up per page.
DROP TABLE IF EXISTS _pi_slots;
CREATE TABLE _pi_slots (plus_id INT PRIMARY KEY, i0 TEXT, i1 TEXT, i2 TEXT, t0 MEDIUMTEXT, t1 MEDIUMTEXT, t2 MEDIUMTEXT, t3 MEDIUMTEXT) ENGINE=InnoDB
SELECT pp.plus_id, img.i0, img.i1, img.i2, txt.t0, txt.t1, txt.t2, txt.t3
FROM _pi_pages pp
LEFT JOIN (SELECT page_id, MAX(IF(slot = 0, image, NULL)) AS i0, MAX(IF(slot = 1, image, NULL)) AS i1, MAX(IF(slot = 2, image, NULL)) AS i2
           FROM catalog_page_images GROUP BY page_id) img ON img.page_id = pp.plus_id
LEFT JOIN (SELECT page_id, MAX(IF(slot = 0, text, NULL)) AS t0, MAX(IF(slot = 1, text, NULL)) AS t1, MAX(IF(slot = 2, text, NULL)) AS t2,
                  MAX(IF(slot = 3, text, NULL)) AS t3
           FROM catalog_page_texts GROUP BY page_id) txt ON txt.page_id = pp.plus_id;

DROP TABLE IF EXISTS _pi_page_text;
CREATE TABLE _pi_page_text (polaris_id INT PRIMARY KEY, headline TEXT, teaser TEXT, special TEXT, text1 MEDIUMTEXT, text2 MEDIUMTEXT, details MEDIUMTEXT, text_teaser MEDIUMTEXT) ENGINE=InnoDB
SELECT pp.polaris_id, s.i0 AS headline, s.i1 AS teaser, s.i2 AS special, s.t0 AS text1,
  CASE WHEN pp.layout IN ('frontpage', 'pets', 'pets2', 'pets3', 'productpage1', 'roomads', 'mad_money') THEN s.t1
       WHEN pp.layout = 'bots' THEN s.t2
       WHEN pp.layout = 'single_bundle' THEN s.t3 END AS text2,
  CASE WHEN pp.layout IN ('pets', 'pets2', 'pets3', 'productpage1') THEN s.t2
       WHEN pp.layout IN ('frontpage', 'roomads', 'mad_money', 'club_buy', 'club_gift', 'info_duckets', 'info_loyalty', 'info_pets') THEN NULL
       ELSE s.t1 END AS details,
  CASE WHEN pp.layout = 'frontpage' THEN s.t2
       WHEN pp.layout IN ('pets', 'pets2', 'pets3', 'productpage1') THEN s.t3
       WHEN pp.layout IN ('bots', 'roomads', 'mad_money', 'soundmachine', 'club_buy', 'club_gift', 'info_duckets', 'info_loyalty') THEN NULL

DROP TABLE IF EXISTS _pi_products;
CREATE TABLE _pi_products (offer_id INT, position INT, item_id INT UNSIGNED NULL, amount INT, param VARCHAR(1024), kind VARCHAR(20), PRIMARY KEY (offer_id, position)) ENGINE=InnoDB
SELECT op.offer_id, op.position, op.amount, op.product_type AS kind,
  CASE op.product_type WHEN 'furni' THEN fm.polaris_id ELSE sp.item_id END AS item_id,
  CASE op.product_type
    -- For an effect Polaris reads extradata as its duration in seconds (empty: one day), as Plus does.
    WHEN 'bot' THEN CONCAT('name:', bp.name, ';motto:', bp.motto, ';figure:', bp.figure, ';gender:', bp.gender)
    -- Wallpaper, floor and landscape patterns come from the catalog name in Polaris.
    WHEN 'furni' THEN IF(f.interaction_type IN ('wallpaper', 'floor', 'landscape'), '', op.extra_param)
    ELSE op.extra_param
  END AS param
FROM catalog_offer_products op
LEFT JOIN _pi_furni fm ON op.product_type = 'furni' AND fm.plus_id = op.furniture_id
LEFT JOIN furniture f ON op.product_type = 'furni' AND f.id = op.furniture_id
LEFT JOIN catalog_bot_presets bp ON op.product_type = 'bot' AND bp.id = op.bot_preset_id
LEFT JOIN _pi_special sp ON sp.kind = op.product_type COLLATE utf8mb4_general_ci AND sp.product_key = CASE op.product_type
    WHEN 'effect' THEN CAST(op.effect_id AS CHAR) WHEN 'badge' THEN op.badge_code WHEN 'pet' THEN CAST(op.pet_type AS CHAR)
    WHEN 'bot' THEN CAST(op.bot_preset_id AS CHAR) END COLLATE utf8mb4_general_ci;

-- Per offer once: its items as Polaris writes them ("id", or "id;id:amount" for a bundle), amount and extradata.
DROP TABLE IF EXISTS _pi_offer_items;
CREATE TABLE _pi_offer_items (offer_id INT PRIMARY KEY, item_ids VARCHAR(1000), amount INT, param VARCHAR(1024), products INT) ENGINE=InnoDB
SELECT x.offer_id,
  GROUP_CONCAT(IF(x.amount > 1 AND n.products > 1, CONCAT(x.item_id, ':', x.amount), x.item_id) ORDER BY x.position SEPARATOR ';') AS item_ids,
  IF(n.products = 1, MAX(x.amount), 1) AS amount,
  SUBSTRING_INDEX(GROUP_CONCAT(NULLIF(x.param, '') ORDER BY x.position SEPARATOR '|~|'), '|~|', 1) AS param,
  n.products
FROM _pi_products x
JOIN (SELECT offer_id, COUNT(*) AS products FROM _pi_products WHERE item_id IS NOT NULL GROUP BY offer_id) n ON n.offer_id = x.offer_id
WHERE x.item_id IS NOT NULL
GROUP BY x.offer_id, n.products;
